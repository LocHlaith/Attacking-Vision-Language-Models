六、复现统计命令

PowerShell:

$files = rg --files -g '*.py' -g '!outputs/**' -g '!slides/**' -g '!models/**' `
  -g '!figs/**' -g '!datasets/**' -g '!papers/**' -g '!.venv/**' `
  -g '!.git/**' -g '!.pytest_cache/**' -g '!.ruff_cache/**'
$rows = foreach ($f in $files) {
  [pscustomobject]@{
    Path = $f
    Lines = ([System.IO.File]::ReadAllLines((Resolve-Path -LiteralPath $f))).Count
  }
}
$rows | Sort-Object Path
($rows | Measure-Object Lines -Sum).Sum
